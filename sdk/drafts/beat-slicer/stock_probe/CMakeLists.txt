cmake_minimum_required(VERSION 3.16)
project(beat_slicer_stock_probe CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT OCTABAM_ROOT OR NOT OT_VENDOR)
  message(FATAL_ERROR "Supply the pinned upstream OCTABAM_ROOT and OT_VENDOR; see TESTING.md")
endif()
add_subdirectory("${OCTABAM_ROOT}/tools/emu/ot_emu" emulator)
add_executable(beat_slicer_stock_probe probe.cpp)
target_link_libraries(beat_slicer_stock_probe PRIVATE ot_machine)
target_compile_options(beat_slicer_stock_probe PRIVATE -Wall -Wextra -Werror)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/probe.cpp" STOCK_PROBE_SHA)
target_compile_definitions(beat_slicer_stock_probe PRIVATE PROBE_SOURCE_SHA="${STOCK_PROBE_SHA}")
add_executable(beat_slicer_planner_probe planner_probe.cpp)
target_link_libraries(beat_slicer_planner_probe PRIVATE ot_machine)
target_compile_options(beat_slicer_planner_probe PRIVATE -Wall -Wextra -Werror)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/planner_probe.cpp" PLANNER_PROBE_SHA)
target_compile_definitions(beat_slicer_planner_probe PRIVATE PROBE_SOURCE_SHA="${PLANNER_PROBE_SHA}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS probe.cpp planner_probe.cpp)
add_executable(beat_slicer_job_probe job_probe.cpp)
target_link_libraries(beat_slicer_job_probe PRIVATE ot_machine)
target_compile_options(beat_slicer_job_probe PRIVATE -Wall -Wextra -Werror)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/job_probe.cpp" JOB_PROBE_SHA)
target_compile_definitions(beat_slicer_job_probe PRIVATE PROBE_SOURCE_SHA="${JOB_PROBE_SHA}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS job_probe.cpp)
