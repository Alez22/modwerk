cmake_minimum_required(VERSION 3.16)
project(beat_slicer_stock_probe CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT OCTABAM_ROOT OR NOT OT_VENDOR)
  message(FATAL_ERROR "Supply the pinned upstream OCTABAM_ROOT and OT_VENDOR; see TESTING.md")
endif()
add_subdirectory("${OCTABAM_ROOT}/tools/emu/ot_emu" emulator)
add_executable(beat_slicer_stock_probe probe.cpp)

target_compile_options(beat_slicer_stock_probe PRIVATE -Wall -Wextra -Werror)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/probe.cpp" STOCK_PROBE_SHA)
target_compile_definitions(beat_slicer_stock_probe PRIVATE PROBE_SOURCE_SHA="${STOCK_PROBE_SHA}")

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS probe.cpp planner_probe.cpp)

target_compile_options(beat_slicer_coherence_probe PRIVATE -Wall -Wextra -Werror)
# Bind all original C/C++ inputs, not only the runner, into this mixed probe.
set(COHERENCE_INPUTS coherence_probe.cpp ../planner.c ../planner.h ../planner_job.c ../planner_job.h ../pcm_reader.c ../pcm_reader.h)
set(COHERENCE_BINDING "")
foreach(INPUT IN LISTS COHERENCE_INPUTS)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${INPUT}" INPUT_SHA)
  string(APPEND COHERENCE_BINDING "${INPUT}:${INPUT_SHA}\n")
endforeach()
string(SHA256 COHERENCE_PROBE_SHA "${COHERENCE_BINDING}")
target_compile_definitions(beat_slicer_coherence_probe PRIVATE PROBE_SOURCE_SHA="${COHERENCE_PROBE_SHA}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${COHERENCE_INPUTS})
